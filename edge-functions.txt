// supabase functions deploy admin-account-lifecycle
//
// POST { action: 'reactivate' | 'purge', userId: string, note?: string }
//   reactivate → active superadmin or support_admin
//   purge      → active superadmin only (only works on user-deleted accounts)
//
// Optional secret RESERVED_EMAIL_RETENTION_DAYS (set from your retention policy):
//   unset → email stays reserved indefinitely after a purge (safe default)
//   0     → released immediately;  N → released N days after the purge
import { createClient } from 'npm:@supabase/supabase-js@2';

const SUPABASE_URL = Deno.env.get('SUPABASE_URL')!;
const SERVICE_KEY = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!;
const RETENTION_RAW = Deno.env.get('RESERVED_EMAIL_RETENTION_DAYS');
const RETENTION_DAYS =
  RETENTION_RAW && /^\d+$/.test(RETENTION_RAW) ? Number(RETENTION_RAW) : null;

const cors = {
  'Access-Control-Allow-Origin': '*', // tighten to your admin panel's origin
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};
const json = (status: number, body: Record<string, unknown>) =>
  new Response(JSON.stringify(body), {
    status,
    headers: { ...cors, 'Content-Type': 'application/json' },
  });

const UUID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
const KNOWN_DB_ERRORS: Record<string, number> = {
  ACCOUNT_NOT_FOUND: 404,
  NOT_A_DELETED_ACCOUNT: 409,
  ONLY_DELETED_ACCOUNTS_CAN_BE_PURGED: 409,
  WALLET_BALANCE_REMAINING: 409,
};

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: cors });
  if (req.method !== 'POST') return json(405, { code: 'BAD_REQUEST' });

  const jwt = req.headers.get('Authorization')?.replace(/^Bearer\s+/i, '');
  if (!jwt) return json(401, { code: 'UNAUTHENTICATED' });

  const admin = createClient(SUPABASE_URL, SERVICE_KEY, {
    auth: { persistSession: false, autoRefreshToken: false },
  });

  const { data: callerData, error: callerErr } = await admin.auth.getUser(jwt);
  const caller = callerData?.user;
  if (callerErr || !caller) return json(401, { code: 'UNAUTHENTICATED' });

  let body: { action?: unknown; userId?: unknown; note?: unknown };
  try {
    body = await req.json();
  } catch {
    return json(400, { code: 'BAD_REQUEST' });
  }

  const action = body.action === 'reactivate' || body.action === 'purge' ? body.action : null;
  const userId = typeof body.userId === 'string' && UUID.test(body.userId) ? body.userId : null;
  const note = typeof body.note === 'string' ? body.note.trim().slice(0, 1000) : null;
  if (!action || !userId) return json(400, { code: 'BAD_REQUEST' });
  if (userId === caller.id) return json(400, { code: 'CANNOT_TARGET_SELF' });

  const { data: allowed, error: roleErr } = await admin.rpc('is_account_admin', {
    p_user_id: caller.id,
    p_superadmin_only: action === 'purge',
  });
  if (roleErr || allowed !== true) return json(403, { code: 'FORBIDDEN' });

  // Best effort: a logging failure must not undo or block the action itself.
  const audit = async (meta: Record<string, unknown>) => {
    const { error } = await admin.from('admin_audit_log').insert({
      admin_id: caller.id, // if your panel logs admins.id instead, look it up and use that
      admin_email: caller.email ?? null,
      action: `account.${action}`,
      target_type: 'user',
      target_id: userId,
      meta: { note, ...meta },
      ip: req.headers.get('x-forwarded-for')?.split(',')[0]?.trim() ?? null,
      user_agent: req.headers.get('user-agent'),
    });
    if (error) console.error('admin_audit_log insert failed', error);
  };

  try {
    if (action === 'reactivate') {
      const { data: status, error } = await admin.rpc('admin_reactivate_deleted_account', {
        p_user_id: userId,
      });
      if (error) throw error;

      // DB first, then lift the ban. A failed unban is safe to retry ('already_active').
      const { error: unbanErr } = await admin.auth.admin.updateUserById(userId, {
        ban_duration: 'none',
      });
      if (unbanErr) throw unbanErr;

      await audit({ result: status });
      return json(200, { status });
    }

    // purge
    const { data: email, error: prepErr } = await admin.rpc('admin_prepare_purge', {
      p_user_id: userId,
      p_retain_days: RETENTION_DAYS,
    });
    if (prepErr) throw prepErr;

    // Fails with "Database error deleting user" if any table references the user
    // with ON DELETE RESTRICT / NO ACTION. See the FK query in the setup notes.
    const { error: delErr } = await admin.auth.admin.deleteUser(userId);
    if (delErr) throw delErr;

    await audit({ email, email_retention_days: RETENTION_DAYS });
    return json(200, { status: 'purged', emailRetentionDays: RETENTION_DAYS });
  } catch (e) {
    const message =
      e instanceof Error ? e.message : String((e as { message?: string })?.message ?? e);
    const known = Object.keys(KNOWN_DB_ERRORS).find((k) => message.includes(k));
    if (known) return json(KNOWN_DB_ERRORS[known], { code: known });
    console.error('admin-account-lifecycle failed', action, userId, e);
    return json(500, { code: 'SERVER_ERROR', message });
  }
});

import { createClient, SupabaseClient } from 'npm:@supabase/supabase-js@2';

const SUPABASE_URL = Deno.env.get('SUPABASE_URL')!;
const ANON_KEY = Deno.env.get('SUPABASE_ANON_KEY')!;
const SERVICE_KEY = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!;

// ~100 years. Supabase has no permanent ban; reactivation sets ban_duration 'none'.
const PERMANENT_BAN = '876000h';
const NO_SESSION = { auth: { persistSession: false, autoRefreshToken: false } };

const cors = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

const json = (status: number, body: Record<string, unknown>) =>
  new Response(JSON.stringify(body), {
    status,
    headers: { ...cors, 'Content-Type': 'application/json' },
  });

/** Blocks sign-in and token refresh, and revokes every existing refresh token. */
async function revokeAccess(admin: SupabaseClient, userId: string, jwt: string) {
  const { error: banErr } = await admin.auth.admin.updateUserById(userId, {
    ban_duration: PERMANENT_BAN,
  });
  if (banErr) throw banErr;
  // Best effort: the ban already stops refreshes; this kills sessions on other devices sooner.
  await admin.auth.admin.signOut(jwt, 'global').catch(() => undefined);
}

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: cors });
  if (req.method !== 'POST') return json(405, { code: 'BAD_REQUEST', message: 'Method not allowed.' });

  const jwt = req.headers.get('Authorization')?.replace(/^Bearer\s+/i, '');
  if (!jwt) return json(401, { code: 'UNAUTHENTICATED', message: 'Sign in again and retry.' });

  const admin = createClient(SUPABASE_URL, SERVICE_KEY, NO_SESSION);

  const { data: userData, error: userErr } = await admin.auth.getUser(jwt);
  const user = userData?.user;
  if (userErr || !user?.email) {
    return json(401, { code: 'UNAUTHENTICATED', message: 'Sign in again and retry.' });
  }

  let body: { mode?: unknown; password?: unknown; reason?: unknown };
  try {
    body = await req.json();
  } catch {
    return json(400, { code: 'BAD_REQUEST', message: 'Invalid request.' });
  }

  const mode = body.mode === 'check' || body.mode === 'delete' ? body.mode : null;
  const password = typeof body.password === 'string' ? body.password : '';
  const reason = typeof body.reason === 'string' ? body.reason.trim().slice(0, 1000) : '';
  if (!mode || !password || !reason) {
    return json(400, { code: 'BAD_REQUEST', message: 'Enter your password and a reason, then try again.' });
  }

  try {
    const { data: profile, error: profileErr } = await admin
      .from('profiles')
      .select('account_status')
      .eq('id', user.id)
      .single();
    if (profileErr) throw profileErr;

    // Idempotent retry: DB already says deleted (e.g. the ban call failed last time).
    if (profile.account_status === 'deleted') {
      if (mode === 'delete') await revokeAccess(admin, user.id, jwt);
      return json(200, { status: 'deleted' });
    }
    if (profile.account_status !== 'active') {
      return json(403, {
        code: 'ACCOUNT_NOT_ACTIVE',
        message: 'This account cannot be deleted right now. Please contact Support.',
      });
    }

    // Verify the password on a throwaway client so the caller's session is untouched.
    const verifier = createClient(SUPABASE_URL, ANON_KEY, NO_SESSION);
    const { data: signIn, error: pwErr } = await verifier.auth.signInWithPassword({
      email: user.email,
      password,
    });
    if (pwErr?.status === 429) {
      return json(429, { code: 'RATE_LIMITED', message: 'Too many attempts. Wait a few minutes and try again.' });
    }
    if (pwErr || signIn.user?.id !== user.id) {
      return json(401, { code: 'INVALID_PASSWORD', message: 'The password you entered is incorrect.' });
    }
    await verifier.auth.signOut({ scope: 'local' }).catch(() => undefined);

    if (mode === 'check') {
      const { data: blockers, error } = await admin.rpc('account_deletion_blockers', {
        p_user_id: user.id,
      });
      if (error) throw error;
      if (Array.isArray(blockers) && blockers.length > 0) {
        return json(409, { code: 'ACCOUNT_DELETION_BLOCKED', blockers });
      }
      return json(200, { status: 'eligible' });
    }

    // Authoritative path: re-checks everything under row locks in one transaction.
    const { data: result, error } = await admin.rpc('request_account_deletion', {
      p_user_id: user.id,
      p_reason: reason,
    });
    if (error) throw error;

    switch (result?.status) {
      case 'blocked':
        return json(409, { code: 'ACCOUNT_DELETION_BLOCKED', blockers: result.blockers ?? [] });
      case 'not_active':
        return json(403, {
          code: 'ACCOUNT_NOT_ACTIVE',
          message: 'This account cannot be deleted right now. Please contact Support.',
        });
      case 'deleted':
      case 'already_deleted':
        await revokeAccess(admin, user.id, jwt);
        return json(200, { status: 'deleted' });
      default:
        throw new Error(`Unexpected result: ${JSON.stringify(result)}`);
    }
  } catch (e) {
    console.error('delete-account failed', user.id, e);
    return json(500, { code: 'SERVER_ERROR', message: 'We could not delete your account. Please try again.' });
  }
});

      .from('wallet_transactions')
      .update({ status: 'success', flw_ref: flwRef })
      .eq('id', txn.id);

    await admin.rpc('increment_wallet_balance', {
      p_user_id: txn.user_id,
      p_amount: txn.amount_kobo,
    });

    return new Response('OK', { status: 200 });
  }

  // Handle a WITHDRAWAL payout finishing (money going OUT of the wallet)
  if (payload.event === 'transfer.completed') {
    const txRef = payload.data.reference;
    const success = payload.data.status === 'SUCCESSFUL';

      .from('withdrawal_requests')

      });

    return new Response('OK', { status: 200 });
  }

  // Anything else Flutterwave sends, ignore it
  return new Response('Ignored', { status: 200 });

type NotificationCategory =
  | 'order_updates'
  | 'wallet_credits'
  | 'network_updates'
  | 'promotions'
  | 'third_party_updates'
  | 'eplan_subscription_updates'
  | 'ewash_subscription_updates'
  | 'chat_messages'
  | 'welcome';

type NotificationRecord = {
  id: string;
  user_id: string;
  category: NotificationCategory;
  title: string;
  body: string;
  data: Record<string, unknown> | null;
};

const CHAT_REPLY_MESSAGE =
  'Support has replied to your live chat. Open the Hemera app to continue the conversation.';
const categories = new Set<NotificationCategory>([
  'order_updates',
  'wallet_credits',
  'network_updates',
  'promotions',
  'third_party_updates',
  'eplan_subscription_updates',
  'ewash_subscription_updates',
  'chat_messages',
  'welcome',
]);

const settingForCategory: Partial<Record<NotificationCategory, string>> = {
  order_updates: 'order_updates',
  wallet_credits: 'wallet_credits',
  network_updates: 'network_updates',
  promotions: 'promotions',

  eplan_subscription_updates: 'eplan_subscription_updates',
  ewash_subscription_updates: 'ewash_subscription_updates',
  chat_messages: 'chat_messages',

function escapeHtml(value: string): string {
  return value
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#39;');
}

async function sendExpoPush(
  tokens: string[],
  notification: NotificationRecord,
  title: string,
  body: string
) {
  for (let index = 0; index < tokens.length; index += 100) {
    const batch = tokens.slice(index, index + 100);
    const response = await fetch('https://exp.host/--/api/v2/push/send', {
      method: 'POST',
      headers: {
        Accept: 'application/json',
        'Accept-encoding': 'gzip, deflate',
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(
        batch.map((to) => ({
          to,
          title,
          body,
          sound: 'default',
          channelId: 'default',
          data: { notificationId: notification.id, category: notification.category },
        }))
      ),
    });

    if (!response.ok) {
      throw new Error(`Expo push service returned HTTP ${response.status}.`);
    }

    const result = await response.json();
    const tickets = Array.isArray(result.data) ? result.data : [];
    const failedTicket = tickets.find(
      (ticket: { status?: string; message?: string }) => ticket.status === 'error'
    );
    if (failedTicket) {
      throw new Error(`Expo push delivery failed: ${failedTicket.message ?? 'unknown push error'}`);
    }
  }
}

async function sendBrevoEmail(to: string, title: string, body: string) {
  const apiKey = Deno.env.get('BREVO_API_KEY');
  const senderEmail = Deno.env.get('BREVO_SENDER_EMAIL');
  const senderName = Deno.env.get('BREVO_SENDER_NAME') ?? 'Hemera';
  if (!apiKey || !senderEmail) {
    throw new Error('BREVO_API_KEY and BREVO_SENDER_EMAIL must be configured to send notification email.');
  }

  const response = await fetch('https://api.brevo.com/v3/smtp/email', {
    method: 'POST',
    headers: {
      accept: 'application/json',
      'api-key': apiKey,
      'content-type': 'application/json',
    },
    body: JSON.stringify({
      sender: { name: senderName, email: senderEmail },
      to: [{ email: to }],
      subject: title,
      textContent: body,
      htmlContent: `<html><body><p>${escapeHtml(body).replaceAll('\n', '<br />')}</p></body></html>`,
    }),
  });

  if (!response.ok) {
    throw new Error(`Brevo email service returned HTTP ${response.status}.`);
  }
}

Deno.serve(async (request) => {
  if (request.method !== 'POST') return new Response('Method not allowed', { status: 405 });

  const dispatchSecret = Deno.env.get('NOTIFICATION_DISPATCH_SECRET');
  if (!dispatchSecret) {
    console.error('[Notifications] NOTIFICATION_DISPATCH_SECRET is not configured.');
    return new Response('Notification delivery is not configured', { status: 503 });
  }
  if (request.headers.get('x-notification-secret') !== dispatchSecret) {
    return new Response('Unauthorized', { status: 401 });
  }

  const payload = await request.json();
  const record = payload?.record as NotificationRecord | undefined;
  if (
    !record ||
    !record.id ||
    !record.user_id ||
    !record.title ||
    !categories.has(record.category)
  ) {
    return new Response('Invalid notification record', { status: 400 });
  }

  const supabaseUrl = Deno.env.get('SUPABASE_URL');
  const serviceRoleKey = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY');
  if (!supabaseUrl || !serviceRoleKey) {
    console.error('[Notifications] Supabase service credentials are not configured.');
    return new Response('Notification delivery is not configured', { status: 503 });
  }

  const admin = createClient(supabaseUrl, serviceRoleKey);
  const [
    { data: settings, error: settingsError },
    { data: tokenRows, error: tokensError },
    { data: userResult, error: userError },
  ] = await Promise.all([
    admin.from('notification_settings').select('*').eq('user_id', record.user_id).maybeSingle(),
    admin.from('notification_push_tokens').select('expo_push_token').eq('user_id', record.user_id),
    admin.auth.admin.getUserById(record.user_id),
  ]);

  if (settingsError || tokensError || userError) {
    const error = settingsError ?? tokensError ?? userError;
    console.error('[Notifications] Could not load notification delivery data:', error);
    return new Response('Could not load notification delivery data', { status: 500 });
  }

  const preference = settingForCategory[record.category];
  if (preference && settings?.[preference] === false) {
    return Response.json({ skipped: 'category_disabled' });
  }

  const title = record.category === 'chat_messages' ? 'New live chat reply' : record.title;
  const body = record.category === 'chat_messages' ? CHAT_REPLY_MESSAGE : record.body;
  const deliveries: Promise<void>[] = [];

  if (settings?.push_enabled !== false && tokenRows?.length) {
    deliveries.push(
      sendExpoPush(
        tokenRows.map((token: { expo_push_token: string }) => token.expo_push_token),
        record,
        title,
        body
      )
    );
  }

  if (settings?.email_enabled !== false && userResult.user.email) {
    deliveries.push(sendBrevoEmail(userResult.user.email, title, body));
  }

  const results = await Promise.allSettled(deliveries);
  const failures = results
    .filter((result): result is PromiseRejectedResult => result.status === 'rejected')
    .map((result) => result.reason instanceof Error ? result.reason.message : 'Unknown delivery error');

    console.error('[Notifications] One or more notification channels failed:', failures);

  });

import { createClient } from 'jsr:@supabase/supabase-js@2';

Deno.serve(async (req) => {
  const authHeader = req.headers.get('Authorization')!;
  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!,
    { global: { headers: { Authorization: authHeader } } }
  );

  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return new Response('Unauthorized', { status: 401 });

  const { requestId } = await req.json();
  if (!requestId) return new Response('Missing requestId', { status: 400 });

  const admin = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  );

  const { data: request, error: fetchError } = await admin
    .from('withdrawal_requests')
    .select('id, user_id, amount_kobo, tx_ref, status, bank_account_id')
    .eq('id', requestId)
    .eq('user_id', user.id)
    .single();

  if (fetchError || !request) {
    return new Response(JSON.stringify({ error: 'Withdrawal request not found' }), { status: 404 });
  }
  if (request.status !== 'pending') {
    return new Response(JSON.stringify({ error: 'already_processed' }), { status: 200 });
  }

  const { data: bankAccount } = await admin
    .from('bank_accounts')
    .select('bank_code, account_number, account_name')
    .eq('id', request.bank_account_id)
    .single();

  if (!bankAccount) {
    return new Response(JSON.stringify({ error: 'Bank account not found' }), { status: 404 });
  }

  const flwRes = await fetch('https://api.flutterwave.com/v3/transfers', {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${Deno.env.get('FLW_SECRET_KEY')}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      account_bank: bankAccount.bank_code,
      account_number: bankAccount.account_number,
      amount: Math.round(request.amount_kobo / 100),
      currency: 'NGN',
      reference: request.tx_ref,
      narration: `Hemera Withdrawal - ${bankAccount.account_name}`,
    }),
  });

  const flwData = await flwRes.json();
  console.log('flutterwave transfer response:', JSON.stringify(flwData));

  if (flwData.status !== 'success') {
    await admin.from('withdrawal_requests').update({ status: 'failed' }).eq('id', requestId);
    await admin.rpc('increment_wallet_balance', { p_user_id: user.id, p_amount: request.amount_kobo });
    await admin.from('wallet_transactions').update({ status: 'failed' }).eq('tx_ref', request.tx_ref);
    return new Response(JSON.stringify({ error: flwData.message ?? 'Transfer failed' }), { status: 502 });
  }

  await admin
    .from('withdrawal_requests')
    .update({ status: 'processing', flw_transfer_id: String(flwData.data.id) })
    .eq('id', requestId);

  return new Response(JSON.stringify({ status: 'processing' }), { status: 200 });
});
